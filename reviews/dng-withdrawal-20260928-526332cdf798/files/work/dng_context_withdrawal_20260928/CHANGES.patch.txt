+def withdrawal(operands, mask, sign, drive, theta, warp):
+    """Zero reception of a predefined signed subset, with all other operands fixed."""
+    w, q, cap, included = validate_operands(operands, mask)
+    need(sign in {'none', 'negative', 'positive', 'all'}, 'unknown withdrawal sign')
+    selected = np.broadcast_to(mask, q.shape).copy()
+    if sign == 'none': selected[:] = False
+    elif sign == 'negative': selected &= w < 0
+    elif sign == 'positive': selected &= w > 0
+    terms = np.where(included != 0, w*(q*cap), np.float32(0))
+    after_q = np.where(selected, np.float32(0), q)
+    after = np.where(included != 0, w*(after_q*cap), np.float32(0))
+    net, after_net = warp(terms), warp(after)
+    margin = np.float32(net+np.asarray(drive,np.float32))-np.asarray(theta,np.float32)
+    after_margin = np.float32(after_net+np.asarray(drive,np.float32))-np.asarray(theta,np.float32)
+    need(np.isfinite(after_margin).all(), 'nonfinite withdrawal')
+    if sign == 'none': need(np.array_equal(net, after_net), 'identity changed consumer')
+    if sign == 'negative': need(np.all(after_margin >= margin), 'negative withdrawal direction')
+    if sign == 'positive': need(np.all(after_margin <= margin), 'positive withdrawal direction')
+    return dict(original_net32=net, original_margin32=margin, after_net32=after_net,
+                after_margin32=after_margin, increment32=after_net.astype(float)-net,
+                removed_product_sum64=np.where(selected,terms,np.float32(0)).sum(axis=-1,dtype=np.float64))
